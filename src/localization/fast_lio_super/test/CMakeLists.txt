cmake_minimum_required(VERSION 3.16)
project(fastlio_contract_checks LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PCL 1.8 REQUIRED COMPONENTS common filters registration)
find_package(GTSAM 4.2 REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenMP)
include_directories(../include ../../slam_octvox/include ${PCL_INCLUDE_DIRS} ${GTSAM_INCLUDE_DIR})
add_library(map_core ../src/slam/map/ikd_tree_map.cpp ../src/slam/map/oct_vox_map.cpp
  ../include/ikd-Tree/ikd_Tree.cpp)
target_link_libraries(map_core PUBLIC ${PCL_LIBRARIES} Threads::Threads)
add_library(graph_core ../src/slam/graph/scan_context.cpp ../src/slam/graph/gicp.cpp
  ../src/slam/graph/pose_graph.cpp ../src/slam/graph/backend.cpp)
target_link_libraries(graph_core PUBLIC gtsam ${PCL_LIBRARIES} Threads::Threads)
enable_testing()
add_executable(map_contract_test map_contract_test.cpp)
target_link_libraries(map_contract_test map_core)
target_compile_options(map_contract_test PRIVATE -UNDEBUG)
add_executable(graph_contract_test graph_contract_test.cpp)
target_link_libraries(graph_contract_test graph_core)
target_compile_options(graph_contract_test PRIVATE -UNDEBUG)
add_test(NAME map_contract COMMAND map_contract_test)
add_test(NAME graph_contract COMMAND graph_contract_test)
