cmake_minimum_required(VERSION 3.16)
project(fast_lio_super_ros2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(PCL REQUIRED COMPONENTS common filters registration)
find_package(Eigen3 REQUIRED)
find_package(GTSAM 4.2 REQUIRED)
find_package(Threads REQUIRED)
if(TARGET metis-gtsam)
  get_target_property(GTSAM_METIS_LIBRARY metis-gtsam IMPORTED_LOCATION_RELEASE)
  if(NOT GTSAM_METIS_LIBRARY)
    get_target_property(GTSAM_METIS_LIBRARY metis-gtsam IMPORTED_LOCATION_NONE)
  endif()
  if(GTSAM_METIS_LIBRARY)
    get_filename_component(GTSAM_METIS_LIBRARY_DIR ${GTSAM_METIS_LIBRARY} DIRECTORY)
    ament_environment_hooks(env-hooks/gtsam_library_path.dsv.in)
  endif()
endif()
foreach(dependency rclcpp message_filters nav_msgs sensor_msgs geometry_msgs diagnostic_msgs pcl_conversions tf2_ros)
  find_package(${dependency} REQUIRED)
endforeach()
set(core ${CMAKE_CURRENT_SOURCE_DIR}/../fast_lio_super)
set(vendor ${CMAKE_CURRENT_SOURCE_DIR}/../slam_octvox)
add_library(fastlio_map SHARED ${core}/src/slam/map/ikd_tree_map.cpp
  ${core}/src/slam/map/oct_vox_map.cpp ${core}/include/ikd-Tree/ikd_Tree.cpp)
target_include_directories(fastlio_map PUBLIC
  $<BUILD_INTERFACE:${core}/include> $<BUILD_INTERFACE:${vendor}/include>
  $<INSTALL_INTERFACE:include> ${PCL_INCLUDE_DIRS})
target_link_libraries(fastlio_map PUBLIC ${PCL_LIBRARIES} Eigen3::Eigen Threads::Threads)
add_library(fastlio_graph SHARED ${core}/src/slam/graph/scan_context.cpp
  ${core}/src/slam/graph/gicp.cpp ${core}/src/slam/graph/pose_graph.cpp ${core}/src/slam/graph/backend.cpp)
target_include_directories(fastlio_graph PUBLIC
  $<BUILD_INTERFACE:${core}/include> $<INSTALL_INTERFACE:include> ${PCL_INCLUDE_DIRS} ${GTSAM_INCLUDE_DIR})
target_link_libraries(fastlio_graph PUBLIC gtsam ${PCL_LIBRARIES} Threads::Threads)
add_executable(fast_lio_super_backend src/backend_node.cpp)
target_link_libraries(fast_lio_super_backend fastlio_graph)
ament_target_dependencies(fast_lio_super_backend rclcpp message_filters nav_msgs sensor_msgs
  geometry_msgs diagnostic_msgs pcl_conversions tf2_ros)
install(TARGETS fastlio_map fastlio_graph EXPORT fastlio_targets
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(TARGETS fast_lio_super_backend DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY ${core}/include/slam ${core}/include/ikd-Tree DESTINATION include
  FILES_MATCHING PATTERN "*.hpp" PATTERN "*.h")
install(DIRECTORY ${vendor}/include/ DESTINATION include)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
ament_export_targets(fastlio_targets HAS_LIBRARY_TARGET)
ament_export_dependencies(PCL Eigen3 GTSAM Threads)
ament_package()
