cmake_minimum_required(VERSION 3.16)
project(greenhouse_mppi_navigation)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
foreach(dep rclcpp nav2_costmap_2d nav2_util pluginlib nav_msgs sensor_msgs geometry_msgs std_msgs
  tf2_ros tf2_sensor_msgs pcl_conversions greenhouse_cloud_normalizer)
  find_package(${dep} REQUIRED)
endforeach()
find_package(PCL REQUIRED COMPONENTS common)
add_library(terrain_layer SHARED src/terrain_layer.cpp)
ament_target_dependencies(terrain_layer rclcpp nav2_costmap_2d nav2_util pluginlib nav_msgs)
pluginlib_export_plugin_description_file(nav2_costmap_2d terrain_plugins.xml)
add_executable(terrain_grid src/terrain_grid.cpp)
ament_target_dependencies(terrain_grid rclcpp sensor_msgs nav_msgs std_msgs geometry_msgs tf2_ros
  tf2_sensor_msgs pcl_conversions greenhouse_cloud_normalizer)
target_include_directories(terrain_grid PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(terrain_grid ${PCL_LIBRARIES})
ament_python_install_package(${PROJECT_NAME})
install(TARGETS terrain_layer LIBRARY DESTINATION lib)
install(TARGETS terrain_grid DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/mission scripts/command_adapter scripts/validate_deployment scripts/identity_odometry DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(navigation_contract test/test_navigation_contract.py TIMEOUT 30)
endif()
ament_package()
